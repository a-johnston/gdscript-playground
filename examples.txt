var spell_part_regex = RegEx.create_from_string("[ZY][ABC]+")

func parse_spell_parts(spell: String) -> Array[String]:
    var len := 0
    var spell_parts: Array[String] = []
    for re_match in spell_part_regex.search_all(spell):
        len += re_match.get_end() - re_match.get_start()
        spell_parts.append(re_match.get_string())
    if len != len(spell):  # Validate we matched the entire string
        # This can be a warning or w/e but this page just does stdout
        prints("bad spell:", spell)
        spell_parts.clear()
    return spell_parts

func _ready():
    for good_spell in ["ZAA", "YBCZAA"]:
		prints(good_spell, parse_spell_parts(good_spell))

    for bad_spell in ["XAA", "AA", "ZZ", "ZAAZ"]:
        parse_spell_parts(bad_spell)

func sort_statuses(a: Status, b: Status):
    if a._weight == b._weight:
        return a._name < b._name

    return a._weight < b._weight

    var _weight := 0
    var _name := ""


    func _init(weight, name):
        _weight = weight
        _name = name
